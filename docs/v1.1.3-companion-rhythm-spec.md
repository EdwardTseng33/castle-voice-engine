# castle/multimodal/pose_hands_dispatch.py (新檔 ~180 行)
import mediapipe as mp
import numpy as np

mp_pose = mp.solutions.pose
mp_hands = mp.solutions.hands

class PoseHandsDispatcher:
    def __init__(self):
        self.pose = mp_pose.Pose(min_detection_confidence=0.6)
        self.hands = mp_hands.Hands(min_detection_confidence=0.6, max_num_hands=2)
        self.state = "present"  # present / absent / waving / chin_rest / hands_up
        self.last_emit_t = 0
        self.cooldown = 4.0  # seconds between emits

    def process(self, frame_rgb):
        pose_res = self.pose.process(frame_rgb)
        hand_res = self.hands.process(frame_rgb)

        new_state = self.classify(pose_res, hand_res)
        now = time.time()
        if new_state != self.state and (now - self.last_emit_t) > self.cooldown:
            self.last_emit_t = now
            self.state = new_state
            return self.map_state_to_signal(new_state)
        return None

    def classify(self, pose_res, hand_res):
        if not pose_res.pose_landmarks:
            return "absent"  # Edward 離開鏡頭
        # 揮手檢測 (hand wrist 高於 nose y)
        if hand_res.multi_hand_landmarks:
            for hand in hand_res.multi_hand_landmarks:
                wrist_y = hand.landmark[0].y
                nose_y = pose_res.pose_landmarks.landmark[0].y
                if wrist_y < nose_y - 0.05:
                    return "waving"
        # 撐臉檢測 (wrist 靠近 chin)
        # ... (省略 · 詳見 implementation)
        return "present"

    def map_state_to_signal(self, state):
        return {
            "absent": {"animation": "task-handoff", "reason": "edward_left_frame"},
            "waving": {"animation": "acknowledgement", "reason": "edward_waved"},
            "chin_rest": {"animation": "acknowledgement", "reason": "edward_thinking"},
            "hands_up": {"animation": "playful", "reason": "edward_excited"},
            "present": None,  # 不 emit、idle 維持
        }.get(state)
```

// 已有 /vision/latest 1Hz poll · 升級判斷邏輯
function handleVisionSignal(payload) {
  if (!payload || !payload.signal) return;

  const sig = payload.signal;
  // 直接呼叫 emit_animation_state（不經 GPT、本地觸發）
  // 因為 pose / hands 反應要即時、不該等 GPT 4-token roundtrip

  if (sig.animation && frequencyGuard.canEmit(sig.animation)) {
    animationPool.play(sig.animation, sig.reason);
  }
}

// poll loop
async function pollVision() {
  const r = await fetch('/vision/latest');
  if (r.ok) handleVisionSignal(await r.json());
}
setInterval(pollVision, 1000);
```
