### 3.4 voice-lifecycle.js（瀏覽器 state machine · 約 100 行）

```javascript
// castle/static/voice-lifecycle.js
class VoiceLifecycle {
  constructor() {
    this.state = "idle";
    this.lastVoiceMs = 0;
    this.lastFaceMs = 0;
    this.musetalkWs = null;
    this.openaiPeer = null;
  }

  tick() {
    const now = Date.now();
    const silenceMs = now - this.lastVoiceMs;
    const faceAbsentMs = now - this.lastFaceMs;

    switch (this.state) {
      case "idle":
        if (this.shouldWake()) this.transitionTo("waking");
        break;

      case "waking":
        if (this.allEndpointsReady()) this.transitionTo("active");
        break;

      case "active":
        if (silenceMs > 30000) this.showHint("我在這、有事再叫");
        if (silenceMs > 60000 && faceAbsentMs > 30000) this.transitionTo("softoff");
        break;

      case "softoff":
        if (silenceMs > 300000) {
          this.transitionTo("idle");
          this.tearDownGPU();
        }
        if (this.shouldWake()) this.transitionTo("active");
        break;
    }
  }

  shouldWake() {
    return this.speechbrainVerified() || (this.faceVisible() && this.faceMs > 3000);
  }

  transitionTo(newState) {
    console.log("[VoiceLifecycle]", this.state, "->", newState);
    if (newState === "idle") {
      fetch("/voice/sleep", { method: "POST", headers: this.authHeaders });
    }
    this.state = newState;
    this.renderUI();
  }

  tearDownGPU() {
    if (this.musetalkWs) this.musetalkWs.close();
    if (this.openaiPeer) this.openaiPeer.close();
  }
}
```
