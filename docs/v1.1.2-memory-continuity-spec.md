### A · IndexedDB schema（browser-side · 純前端、不踩後端）

**新增 `castle/static/conversation-memory.js`** （~150 行）：

```js
// DB: sophie-memory-v1
// Store: conversations (keyPath: sessionId, indexes: [timestamp])
// Store: summaries (keyPath: date, value: { date, summary, mood, promises })

const DB_NAME = 'sophie-memory-v1';
const DB_VERSION = 1;

export async function openDB() {
  return new Promise((resolve, reject) => {
    const req = indexedDB.open(DB_NAME, DB_VERSION);
    req.onupgradeneeded = (e) => {
      const db = e.target.result;
      if (!db.objectStoreNames.contains('conversations')) {
        const store = db.createObjectStore('conversations', { keyPath: 'sessionId' });
        store.createIndex('timestamp', 'timestamp', { unique: false });
      }
      if (!db.objectStoreNames.contains('summaries')) {
        db.createObjectStore('summaries', { keyPath: 'date' });
      }
    };
    req.onsuccess = (e) => resolve(e.target.result);
    req.onerror = (e) => reject(e.target.error);
  });
}

export async function recordTurn(sessionId, role, text) {
  // role: 'user' | 'sophie'
  const db = await openDB();
  const tx = db.transaction('conversations', 'readwrite');
  const store = tx.objectStore('conversations');
  const existing = await new Promise(r => {
    const req = store.get(sessionId);
    req.onsuccess = () => r(req.result || { sessionId, timestamp: Date.now(), turns: [] });
  });
  existing.turns.push({ role, text, t: Date.now() });
  store.put(existing);
}
```

### B · Server endpoint `/memory/summarize`（Claude Haiku · 蘇菲視角摘要）

**新增 `castle/server/memory_endpoints.py`** （~120 行）：

```python
from fastapi import APIRouter, Request
from anthropic import Anthropic
import os

router = APIRouter()
_client = Anthropic(api_key=os.environ.get("ANTHROPIC_API_KEY"))

@router.post("/memory/summarize")
async def summarize(req: Request):
    body = await req.json()
    turns = body.get("turns", [])
    if not turns:
        return {"summary": "", "mood": "neutral", "promises": []}

    transcript = "\n".join(f"{t['role']}: {t['text']}" for t in turns[-30:])

    msg = _client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=400,
        system=(
            "你是蘇菲、Edward 的私人 AI 陪伴。"
            "把這段對話濃縮成你自己的視角紀錄："
            "1. summary (≤ 80 字、第一人稱「我/Edward」、不是「user/assistant」)"
            "2. mood (Edward 當下心情：accomplished/stressed/tired/happy/anxious/calm)"
            "3. promises (蘇菲承諾要做的事 array、沒有就 [])"
            "回 JSON {summary, mood, promises}"
        ),
        messages=[{"role": "user", "content": transcript}]
    )
    return parse_json(msg.content[0].text)
```

**護欄**：
- 不存原對話文字進後端（只進 Anthropic 一次性 API call）
- 摘要回 client、client 寫 IndexedDB
- 對話 raw turns 在 browser 7 天後自動清（IndexedDB TTL job）

### C · 開場使用記憶（pickGreeting 升級 v2）

**index.html**：

```js
async function pickGreetingV2(ctx) {
  // ctx 來自 session-memory.js + conversation-memory.js
  const recent = await getRecentSummaries(7); // 最近 7 天摘要

  if (recent.length === 0) {
    return pickGreeting(ctx); // v1.1.1 fallback
  }

  const lastDay = recent[0]; // 最近一天
  const promises = lastDay.promises || [];

  // 30% 機率用 promise 招呼 (蘇菲記得承諾)
  if (promises.length > 0 && Math.random() < 0.3) {
    return `Edward · 上次說的 ${promises[0]} · 我有記著`;
  }

  // 25% 機率用 mood 招呼
  if (lastDay.mood === 'tired' && Math.random() < 0.25) {
    return `Edward · 昨天看你有點累 · 今天有歇到嗎`;
  }
  if (lastDay.mood === 'accomplished') {
    return `Edward · 接著昨天那股勁`;
  }

  // 否則 fall back v1.1.1
  return pickGreeting(ctx);
}
```
